.replaceAll("&","&amp;")
.replaceAll("<","&lt;")
.replaceAll(">","&gt;")
.replaceAll('"',"&quot;")
.replaceAll("'","&#039;");

.toLowerCase()
.endsWith(".pdf")
);

const hasEPUB =
files.some(file =>
file.name
.toLowerCase()
.endsWith(".epub")
);

"PDF + EPUB";

detectedFormat.textContent =
"PDF + EPUB";

return;

"PDF";

"PDF";

setFormatReady();

return;

currentFormat =
"EPUB";

"EPUB";

function loadSaved(){

const saved =
localStorage.getItem(
"bookly-mvp"
);

if(!saved){

updatePage();

return;

}


try{

const data =
JSON.parse(saved);


fields.title.value =
data.title || "";

fields.author.value =
data.author || "";

fields.tagline.value =
data.tagline || "";

fields.description.value =
data.description || "";

fields.price.value =
data.price || "";

fields.currency.value =
data.currency || "FCFA";

fields.payment.value =
data.payment || "";


coverData =
data.cover || "";


if(coverData){

bookCover.src =
coverData;

bookCover.style.display =
"block";

bookCoverEmpty.style.display =
"none";

coverPreviewMini.src =
coverData;

coverPreviewMini.style.display =
"block";

uploadIcon.style.display =
"none";

}


updatePage();


if(data.format){

currentFormat =
data.format;

detectedFormat.textContent =
data.format;

formatDescription.textContent =
data.format === "PDF"
? "Document PDF disponible."
: data.format === "EPUB"
? "Livre numérique au format EPUB."
: "Deux formats numériques disponibles.";

setFormatReady();

}


saveStatus.textContent =
"Projet chargé";

}
catch(error){

console.warn(
"Impossible de charger le projet."
);

}

}
